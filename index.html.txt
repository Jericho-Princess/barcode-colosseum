  function chaosHash(h){
  h ^= (h << 13);
  h ^= (h >> 17);
  h ^= (h << 5);

  h = (h * 2654435761) >>> 0; // 拡散（黄金比）

  h ^= (h >> 16);
  return h >>> 0;
}

function gen(code=""){

  let atk=40+(hash%15),def=20+(hash%10),hp=300+(hash%70);
  if(rarity=="rare"){atk*=1.15;def*=1.15;hp*=1.15;}
  if(rarity=="ssr"){atk*=1.3;def*=1.3;hp*=1.3;}

  return {
  name:"モンスター",
  rarity,
  sprite,
  hp,
  maxHp:hp,
  atk,
  def,
  tempAtk: atk,
  tempDef: def,
  skills:assignSkillsRandom(rarity),
  status:null
};
}

let codeReader;
let selectedDeviceId;
let lastScan = "";
let lastTime = 0;
let isScanning = false;

async function scan(){
  document.getElementById("stopBtn").style.display = "block";

  if(isScanning) return;
  isScanning = true;

  codeReader = new ZXing.BrowserMultiFormatReader();

  const devices = await codeReader.listVideoInputDevices();
  selectedDeviceId = devices[0].deviceId;

  codeReader.decodeFromVideoDevice(
    selectedDeviceId,
    'video',
    (result, err) => {
      if(!isScanning) return; // ←停止後の暴走防止

      if(result){
        let txt = result.getText();
        let now = Date.now();

        if(txt !== lastScan || now - lastTime > 1500){
          lastScan = txt;
          lastTime = now;

          addChar(txt);
        }
      }
    }
  );
}

function stopScan(){
  if(codeReader){
    codeReader.reset();
    codeReader = null;
  }

  isScanning = false;

  const video = document.getElementById("video");
  if(video.srcObject){
    video.srcObject.getTracks().forEach(track => track.stop());
    video.srcObject = null;
  }

  document.getElementById("stopBtn").style.display = "none";
}

 function createHash(code){
  let h = 0;
  for(let i=0;i<code.length;i++){
    h = (h << 5) - h + code.charCodeAt(i);
    h |= 0;
  }

  // ⭐ ここが重要（カオス化）
  h = chaosHash(h);

  // ⭐ さらに混ぜる（解析対策）
  h ^= code.length * 1234567;
  h ^= code.charCodeAt(0) << 8;
  h ^= code.charCodeAt(code.length-1);

  return h >>> 0;
}

function createCharacterFromBarcode(code){

  let hash = createHash(code);

  // ⭐ 出現率そのまま（4% / 20%）
  let r = (hash % 100) / 100;

  let rarity = "common";
  if(r < 0.04) rarity = "ssr";
  else if(r < 0.24) rarity = "rare";

   let atk = 40 + (hash % 15);
  let def = 20 + (hash % 10);
  let hp  = 300 + (hash % 70);

  if(rarity==="rare"){
    atk *= 1.2;
    def *= 1.2;
    hp  *= 1.2;
  }
  if(rarity==="ssr"){
    atk *= 1.4;
    def *= 1.4;
    hp  *= 1.4;
  }

  atk = Math.floor(atk);
  def = Math.floor(def);
  hp  = Math.floor(hp);

  let list = IMAGES[rarity];
  let index = hash % list.length;
  let sprite = list[index];

  // ⭐ 固定スキル使う
  let skills = assignSkillsFixed(hash, rarity);

  let name = "バーコードモンスター" + code.slice(-3);

  return {
    name: name,
    code: code,
    rarity: rarity,
    atk: atk,
    def: def,
    maxHp: hp,
    hp: hp,
    tempAtk: atk,
    tempDef: def,
    sprite: sprite,
    skills: skills,
    status: null
  };
}

function render(){

  } else {

  }

    let hpP = (hpVisual[i] / c.maxHp) * 100;

    let color = "lime";
    if(hpP < 60) color = "orange";
    if(hpP < 30) color = "red";

    let statusMap = {
      paralyze:"⚡麻痺",
      poison:"☠️猛毒",
      weaken:"🩻衰弱",
      irradiate:"☢️被曝"
    };

    let statusText = "状態：健康";

      statusText = "状態：" + statusMap[c.status];
    }

    html += `
      <div class="char ${c.rarity}">
        <img src="${c.sprite}" style="width:120px;height:160px"><br>

        ❤️ ${Math.floor(c.hp)} / ${Math.floor(c.maxHp)}<br>
        ⚔️ ${Math.floor(c.tempAtk || c.atk)}<br>
        🛡 ${Math.floor(c.tempDef || c.def)}<br>

  });

  document.getElementById("chars").innerHTML = html;

function animateHP(){
  if(animRunning) return;
  animRunning = true;

  function loop(){
    characters.forEach((c,i)=>{
      let target = c.hp;

      hpVisual[i] += (target - hpVisual[i]) * 0.15;

      if(Math.abs(target - hpVisual[i]) < 0.1){
        hpVisual[i] = target;
      }
    });

    render();
    requestAnimationFrame(loop);
  }

  loop();
}

function useSkill(i){
  if(!battleStarted) return;

  // CPU中でもCPUは通す
  if(waitingForCPU && gameMode=="pvp") return;

  let a = characters[current];
  let d = characters[1-current];
  let s = a.skills[i];

  waitingForCPU = true;

  let name = current==0 ? player1NameVal : player2NameVal;
  let enemy = current==0 ? player2NameVal : player1NameVal;

  let msg = name + " が " + s.name + " を使った";

  // 🔥 奥義1回制限
  if(s.name==="奥義" && s.used){
    showEffect("奥義は１回しか使えない ！！");
    waitingForCPU = false;
    return;
  }

  if(s.name==="奥義"){
    s.used = true;
  }

  // 🔥 ミス判定
  if(s.miss && Math.random() < s.miss){
    msg += "<br>Miss ！！";
    render();

    showEffect(msg, ()=>{
      waitingForCPU = false;
      nextTurn();
    });
    return;
  }

  if(s.instant && Math.random() < s.instant){
    d.hp = 0;
    msg += "<br>即死 ！！";
  }

  if(s.heal){
  // ランダム回復
  let rate = 0.07 + Math.random() * 0.08;
  let heal = Math.floor(a.maxHp * rate);

  a.hp += heal;
  if(a.hp > a.maxHp) a.hp = a.maxHp;

  // 防御バフ（忘れがち）
  if(a.tempDef == null) a.tempDef = a.def;
  a.tempDef *= 1.1;

  msg += `<br>${heal} の回復`;
}

  if(a.tempAtk == null) a.tempAtk = a.atk;
if(a.tempDef == null) a.tempDef = a.def;
if(d.tempAtk == null) d.tempAtk = d.atk;
if(d.tempDef == null) d.tempDef = d.def;

// バフ
if(s.buffAtk){
  a.tempAtk *= s.buffAtk;
  msg += "<br>攻撃力アップ！！";
}

if(s.buffDef){
  a.tempDef *= s.buffDef;
  msg += "<br>防御力アップ！！";
}

// デバフ
if(s.debuffAtk){
  d.tempAtk *= s.debuffAtk;
  msg += "<br>相手の攻撃力ダウン！！";
}

if(s.debuffDef){
  d.tempDef *= s.debuffDef;
  msg += "<br>相手の防御力ダウン！！";
}

  if(s.name==="呪詛"){
    d.status = "weaken";
  }

  // 🔥 ダメージ処理
  if(s.power){
  let crit = Math.random() < 0.05;
  let base = ((a.tempAtk || a.atk) - (d.tempDef || d.def)/2) * s.power;

  if(crit){
    base *= 2;
    msg += "<br>Critical ！！";
  }

  let dmg = Math.floor(base);

  // ⭐ 最低1ダメージ保証
  if(dmg < 1) dmg = 1;

  d.hp -= dmg;
  if(d.hp < 0) d.hp = 0;

  msg += "<br>Hit !！ " + dmg + " のダメージ";
  }

  // 🔥 状態異常
  if(s.status && !d.status && Math.random() < (s.rate || 1)){
  d.status = s.status;

  let st = {
    paralyze:"麻痺",
    poison:"猛毒",
    weaken:"衰弱",
    irradiate:"被曝"
  };

  msg += "<br>" + enemy + " は " + st[d.status] + " になった";
}

  render();

  showEffect(msg, ()=>{
  if(d.hp <= 0){
    pendingWin = name; // ←ここで予約
  }

  waitingForCPU = false;
  nextTurn();
});
}

function proceedTurn(){
  if(!battleStarted) return;
  if(waitingForCPU) return;

  let c = characters[current];

  render();
  showSkills();

  if(gameMode=="cpu" && current==1){
    setTimeout(()=>{
      // ⭐ 使えるスキルだけ抽出
      let usable = c.skills.filter(s => !(s.name==="奥義" && s.used));

      let r = Math.floor(Math.random()*usable.length);
      let skill = usable[r];

      let index = c.skills.indexOf(skill);

      useSkill(index);
    },500);
  }
}

function nextTurn(){
  waitingForCPU = false;

  if(pendingWin){
    showWin(pendingWin);
    pendingWin = null;
    return;
  }

  current = 1 - current;

  let c = characters[current];
  let name = current==0 ? player1NameVal : player2NameVal;

  // ⭐ 初期化（永続化）
  if(c.tempAtk == null) c.tempAtk = c.atk;
  if(c.tempDef == null) c.tempDef = c.def;

  // =========================
  // ☠毒
  // =========================
  if(c.status==="poison"){
    let dmg = Math.floor(c.maxHp * 0.04);
    c.hp -= dmg;
    if(c.hp < 0) c.hp = 0;

    render();

    showEffect(name + " は猛毒で " + dmg + "ダメージ !!", ()=>{
      if(c.hp <= 0){
        showWin(current==0 ? player2NameVal : player1NameVal);
        return;
      }
      proceedTurn();
    });
    return;
  }

  // =========================
  // ⚡麻痺
  // =========================
  if(c.status==="paralyze"){
    showEffect(name + " は麻痺で動けない", ()=>{
      c.status = null;
      nextTurn();
    });
    return;
  }

  // =========================
  // 衰弱（修正済み）
  // =========================
  if(c.status==="weaken"){
  let dmg = Math.floor(c.maxHp * 0.04);
  c.hp -= dmg;
  if(c.hp < 0) c.hp = 0;

  // ⭐ これ追加（重要）
  if(c.tempAtk == null) c.tempAtk = c.atk;
  if(c.tempDef == null) c.tempDef = c.def;

  c.tempAtk *= 0.95;
  c.tempDef *= 0.95;

  render();

  showEffect(
    `${name}は衰弱している<br>
     ${dmg} のダメージ！！<br>
     攻撃力と防御力が下がった！！`,
  ()=>{
    if(c.hp <= 0){
      showWin(current==0 ? player2NameVal : player1NameVal);
      return;
    }
    proceedTurn();
  });
  return;
}

  // =========================
  // ☢被曝（修正済み）
  // =========================
  if(c.status==="irradiate"){
  let dmg = Math.floor(c.maxHp * 0.03);
  c.hp -= dmg;
  if(c.hp < 0) c.hp = 0;

  if(c.tempDef == null) c.tempDef = c.def;
  c.tempDef *= 0.8;

  render();

  showEffect(
    `${name}は被曝している<br>
     ${dmg} のダメージ！！<br>
     防御力が下がった！！`,
  ()=>{
    if(c.hp <= 0){
      showWin(current==0 ? player2NameVal : player1NameVal);
      return;
    }
    proceedTurn();
  });
  return;
}

  // =========================
  // 通常ターン
  // =========================
  setTimeout(()=>{
    proceedTurn();
  },50);
}

function showWin(w){
  battleStarted = false;
  waitingForCPU = false;

  document.getElementById("skills").innerHTML = "";
  document.getElementById("turn").innerText = "";

  document.getElementById("rematchBtn").style.display = "inline-block";

  showEffect(`${w} の勝利！！`);
}

function rematch(){
  characters.forEach(c=>{
    c.hp = c.maxHp;
    c.status = null;
    c.tempAtk = c.atk;
    c.tempDef = c.def;

    c.skills.forEach(s=>{
      if(s.name==="奥義") s.used=false;
    });
  });

  hpVisual = characters.map(c=>c.hp);

  current = 0;
  battleStarted = true;
  waitingForCPU = false;
  pendingWin = null;

  document.getElementById("rematchBtn").style.display = "none";

  render();
  showSkills();
  proceedTurn();
}
